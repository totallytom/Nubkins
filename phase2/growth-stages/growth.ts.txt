import { CreatureMood } from '../types';

export type GrowthStage = 'baby' | 'kid' | 'adult';

export const GROWTH_ORDER: GrowthStage[] = ['baby', 'kid', 'adult'];

// Level thresholds. Levels currently cap at 10 (see XP_TABLE in the store).
export function stageForLevel(level: number): GrowthStage {
  if (level >= 8) return 'adult';
  if (level >= 4) return 'kid';
  return 'baby';
}

export interface StageLook {
  label: string;
  scale: number;          // body size relative to the adult art
  prop: string | null;    // shared head prop drawn on any skin (emoji placeholder until art exists)
  bobAmplitude: number;   // idle bounce height (px)
  bobPeriod: number;      // ms per half-bounce — babies bounce quicker
}

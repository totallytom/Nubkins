import React, { useEffect, useRef, useState } from 'react';
import { Animated, Dimensions, Easing, Modal, StyleSheet, Text, TouchableOpacity, View } from 'react-native';
import { FONT, KAWAII, KAWAII_BTN } from '../lib/theme';
import { outfitProps } from '../lib/outfit';
import { GrowthStage, GROWTH_ORDER, STAGE_LOOK } from '../data/growth';
import { getTrait } from '../data/personality';
import { useGameStore } from '../store/useGameStore';
import NubkinCreature from './NubkinCreature';

const { width: SW, height: SH } = Dimensions.get('window');
const SPARKLES = ['✨', '⭐', '💖', '🌟', '💫', '🎉'];
const SPARKLE_N = 16;

interface Props {
  // The stage being grown into; null hides the modal.
  stage: GrowthStage | null;
  onClose: () => void;
}

// The evolution moment: the old form spins and glows, a white flash, then the
// new form pops in with sparkles.
export default function EvolutionModal({ stage, onClose }: Props) {
  const creature = useGameStore(s => s.creature);
  const [revealed, setRevealed] = useState(false);

  const spin    = useRef(new Animated.Value(0)).current;
  const pulse   = useRef(new Animated.Value(1)).current;
  const glow    = useRef(new Animated.Value(0)).current;
  const flash   = useRef(new Animated.Value(0)).current;
  const pop     = useRef(new Animated.Value(0)).current;
  const textIn  = useRef(new Animated.Value(0)).current;
  const sparkles = useRef(
    Array.from({ length: SPARKLE_N }, (_, i) => ({
      emoji: SPARKLES[i % SPARKLES.length],
      angle: (i / SPARKLE_N) * Math.PI * 2,
      dist: 110 + Math.random() * 80,
      t: new Animated.Value(0),
    }))
  ).current;

  useEffect(() => {
    if (!stage) return;
    setRevealed(false);
    [spin, glow, flash, pop, textIn, ...sparkles.map(s => s.t)].forEach(v => v.setValue(0));
    pulse.setValue(1);

    // 1) Old form spins faster and faster while glowing and throbbing.
    Animated.parallel([
      Animated.timing(spin, { toValue: 3, duration: 1800, easing: Easing.in(Easing.cubic), useNativeDriver: true }),
      Animated.timing(glow, { toValue: 1, duration: 1800, useNativeDriver: true }),
      Animated.loop(Animated.sequence([
        Animated.timing(pulse, { toValue: 1.12, duration: 180, useNativeDriver: true }),
        Animated.timing(pulse, { toValue: 0.92, duration: 180, useNativeDriver: true }),
      ]), { iterations: 5 }),
    ]).start(() => {
      // 2) White flash; swap to the new form at its peak.
      Animated.timing(flash, { toValue: 1, duration: 160, useNativeDriver: true }).start(() => {
        setRevealed(true);
        spin.setValue(0);
        pulse.setValue(1);
        // 3) Fade the flash, pop the new form in, burst sparkles, then the text.
        Animated.parallel([
          Animated.timing(flash, { toValue: 0, duration: 650, useNativeDriver: true }),
          Animated.spring(pop, { toValue: 1, speed: 8, bounciness: 16, useNativeDriver: true }),
          Animated.stagger(25, sparkles.map(s =>
            Animated.timing(s.t, { toValue: 1, duration: 1100, easing: Easing.out(Easing.quad), useNativeDriver: true }))),
          Animated.sequence([
            Animated.delay(350),
            Animated.spring(textIn, { toValue: 1, speed: 12, bounciness: 10, useNativeDriver: true }),
          ]),
        ]).start();
      });
    });
  }, [stage]);

  if (!stage) return null;

  const prevStage = GROWTH_ORDER[Math.max(0, GROWTH_ORDER.indexOf(stage) - 1)];
  const shownStage = revealed ? stage : prevStage;
  const outfit = outfitProps(creature.equippedAccessoryId, creature.equippedTattooId, creature.equippedSpecialId);
  const trait = getTrait(creature.trait);

  return (
    <Modal visible transparent animationType="fade" statusBarTranslucent onRequestClose={() => revealed && onClose()}>
      <View style={styles.backdrop}>
        {/* Glow halo behind the Nubkin */}
        <Animated.View style={[styles.halo, { opacity: revealed ? 0.9 : glow, transform: [{ scale: revealed ? 1.15 : pulse }] }]} />

        {/* Sparkle burst on reveal */}
        {revealed && sparkles.map((s, i) => (
          <Animated.Text
            key={i}
            pointerEvents="none"
            style={[styles.sparkle, {
              opacity: s.t.interpolate({ inputRange: [0, 0.7, 1], outputRange: [1, 1, 0] }),
              transform: [
                { translateX: s.t.interpolate({ inputRange: [0, 1], outputRange: [0, Math.cos(s.angle) * s.dist] }) },
                { translateY: s.t.interpolate({ inputRange: [0, 1], outputRange: [0, Math.sin(s.angle) * s.dist] }) },
              ],
            }]}
          >
            {s.emoji}
          </Animated.Text>
        ))}

        <Animated.View
          style={{
            transform: [
              { rotate: spin.interpolate({ inputRange: [0, 3], outputRange: ['0deg', '1080deg'] }) },
              { scale: revealed ? pop.interpolate({ inputRange: [0, 1], outputRange: [0.4, 1.15] }) : pulse },
            ],
          }}
        >
          <NubkinCreature
            mood={revealed ? 'excited' : 'happy'}
            skinId={creature.equippedSkinId}
            {...outfit}
            stage={shownStage}
            trait={creature.trait}
            hideGlow
            onTap={() => {}}
          />
        </Animated.View>

        <Animated.View
          style={[styles.textCard, {
            opacity: textIn,
            transform: [{ translateY: textIn.interpolate({ inputRange: [0, 1], outputRange: [30, 0] }) }],
          }]}
          pointerEvents={revealed ? 'auto' : 'none'}
        >
          <Text style={styles.kicker}>✨ EVOLUTION ✨</Text>
          <Text style={styles.title}>{creature.name} grew into a {STAGE_LOOK[stage].label}!</Text>
          <Text style={styles.sub}>
            All your care is paying off 🥹
            {trait ? `\nStill as ${trait.label.toLowerCase()} as ever ${trait.emoji}` : ''}
          </Text>
          <TouchableOpacity style={styles.btn} onPress={onClose} activeOpacity={0.8}>
            <Text style={styles.btnText}>So proud! 💕</Text>
          </TouchableOpacity>
        </Animated.View>

        {/* White flash at the moment of change */}
        <Animated.View pointerEvents="none" style={[StyleSheet.absoluteFill, styles.flash, { opacity: flash }]} />
      </View>
    </Modal>
  );
}

const styles = StyleSheet.create({
  backdrop: {
    flex: 1,
    backgroundColor: KAWAII.backdrop,
    alignItems: 'center',
    justifyContent: 'center',
  },
  halo: {
    position: 'absolute',
    top: SH / 2 - 190,
    width: 240,
    height: 240,
    borderRadius: 120,
    backgroundColor: '#FFFFFF',
  },
  sparkle: {
    position: 'absolute',
    top: SH / 2 - 110,
    left: SW / 2 - 14,
    fontSize: 28,
  },
  textCard: {
    marginTop: 40,
    width: 320,
    backgroundColor: KAWAII.card,
    borderRadius: 32,
    borderWidth: 5,
    borderColor: KAWAII.cardBorder,
    padding: 20,
    alignItems: 'center',
    gap: 6,
  },
  kicker: { fontFamily: FONT, color: KAWAII.orange, fontSize: 12, fontWeight: '900', letterSpacing: 2 },
  title:  { fontFamily: FONT, color: KAWAII.ink, fontSize: 22, fontWeight: '900', textAlign: 'center' },
  sub:    { fontFamily: FONT, color: KAWAII.inkSoft, fontSize: 13, textAlign: 'center', lineHeight: 19 },
  btn: {
    ...KAWAII_BTN,
    alignSelf: 'stretch',
    alignItems: 'center',
    backgroundColor: KAWAII.pink,
    paddingVertical: 13,
    marginTop: 6,
  },
  btnText: { fontFamily: FONT, color: KAWAII.ink, fontSize: 16, fontWeight: '900' },
  flash:   { backgroundColor: '#FFFFFF' },
});
